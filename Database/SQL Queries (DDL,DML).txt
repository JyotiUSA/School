-- Create the Students table
create table Student
(
StudentID int identity(1,1)  primary key,
Name varchar(50),
Std int,
RollNo int,
BirthDate date
);

-- Create the Exame table
create table Exam
(
	ExameId int primary key,
	StudentID int,
	mathsMarks int,
	languageMarks int,
	socialScienceMarks int,
	scienceMarks int,
	Marks float,
	Percentage float,
	PassOrFail varchar(50),
	Foreign key (StudentID) References Student(StudentID)
);
